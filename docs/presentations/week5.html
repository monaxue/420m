rnorm()

server <- function(input, output, session){

    reset_all <- function() {
    # reset reactives you maintain
    if (exists("pts"))      pts(tibble(x = numeric(), y = numeric()))
    if (exists("mean_now")) mean_now(tibble(Mean = numeric(), y = numeric()))   # if your mean_now is a tibble log
    if (exists("sd_now"))   sd_now(tibble(SD   = numeric(), y = numeric()))       # if your sd_now is a tibble log

    # if in your earlier code mean_now()/sd_now() were scalars, do:
    # if (exists("mean_now")) mean_now(NA_real_)
    # if (exists("sd_now"))   sd_now(NA_real_)

    # clear inputs
    updateTextInput(session, "rnormyours", value = "")
  }

    observeEvent(input$reset, {
    reset_all() })

    output$rnorm1 <- renderText("rnorm(1, mean = 100, sd = 10)")
    output$rnorm25 <- renderText("rnorm(25, mean = 100, sd = 10)")
    output$replicaternorm25 <- renderText("replicate(50, rnorm(25, mean = 100, sd = 10))")

    pts <- reactiveVal(tibble(x = numeric(), y = numeric()))
    mean_now <- reactiveVal(tibble(Mean = numeric(), y = numeric()))
    sd_now <- reactiveVal(tibble(SD = numeric(), y = numeric()))

    observeEvent(input$sample, {
        x_now <- rnorm(1, mean = 100, sd = 10)
        dat <- bind_rows(pts(), tibble(x = x_now)) %>% group_by(round(x)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup() 
        pts(dat)
        }) 

    observeEvent(input$sample25, {
        x_now <- rnorm(25, mean = 100, sd = 10)
        dat <- bind_rows(pts(), tibble(x = x_now)) %>% group_by(round(x)) %>% mutate(y = (row_number()-1) * 0.001)
        pts(dat)

        mean_dat <- mean_now() %>% bind_rows(tibble(Mean = mean(x_now))) %>% group_by(round(Mean)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
        sd_dat <- sd_now() %>% bind_rows(tibble(SD = sd(x_now))) %>% group_by(round(SD)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
        mean_now(mean_dat)
        sd_now(sd_dat)
        }) 

    observeEvent(input$replicate25, {
        x_now <- replicate(50, rnorm(25, mean = 100, sd = 10))
        x_vec <- as.numeric(x_now)
        dat <- bind_rows(pts(), tibble(x = x_vec)) %>% group_by(round(x)) %>% mutate(y = (row_number()-1) * 0.001)
        pts(dat)

        mean_dat <- mean_now() %>% bind_rows(tibble(Mean = as.numeric(colMeans(x_now)))) %>% group_by(round(Mean)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
        sd_dat <- sd_now() %>% bind_rows(tibble(SD = as.numeric(matrixStats::colSds(x_now)))) %>% group_by(round(SD)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
        mean_now(mean_dat)
        sd_now(sd_dat)
        }) 

    observeEvent(input$yoursample, {
        x_now <- tryCatch(eval(parse(text = input$rnormyours)), error = function(e) numeric(0))
        x_vec <- as.numeric(x_now)
        dat <- bind_rows(pts(), tibble(x = x_vec)) %>% group_by(round(x)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
        pts(dat)

        if (length(x_now) > 1 & !is.matrix(x_now)){
          mean_dat <- mean_now() %>% bind_rows(tibble(Mean = mean(x_now))) %>% group_by(round(Mean)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
          sd_dat <- sd_now() %>% bind_rows(tibble(SD = sd(x_now))) %>% group_by(round(SD)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
          mean_now(mean_dat)
          sd_now(sd_dat)
        }
        else if (is.matrix(x_now)){
          mean_dat <- mean_now() %>% bind_rows(tibble(Mean = as.numeric(colMeans(x_now)))) %>% group_by(round(Mean)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
          sd_dat <- sd_now() %>% bind_rows(tibble(SD = as.numeric(matrixStats::colSds(x_now)))) %>% group_by(round(SD)) %>% mutate(y = (row_number()-1) * 0.001) %>% ungroup()
          mean_now(mean_dat)
          sd_now(sd_dat)
        }
        
        
        }) 

    normal = tibble(x = 60:140, Density = dnorm(60:140, mean = 100, sd = 10))
    max_y = max(normal$Density + 0.01)

    output$plot <- renderPlot({ 
      avg = mean(pts()$x)
      sd_all = sd(pts()$x)
      ggplot(normal, aes(x=x, y=Density)) + geom_line(linewidth = 2, color="grey") + theme_classic() + geom_point(data = pts(), aes(x=x,y=y), size=3) + geom_vline(xintercept = avg, color="firebrick") + geom_vline(xintercept = avg+sd_all, color="steelblue") + geom_vline(xintercept = avg-sd_all, color="steelblue") + annotate("label", label = "Mean", x = avg, y =  max_y) + annotate("label", label = "+SD", x = avg+sd_all, y =  max_y) + annotate("label", label = "-SD", x = avg-sd_all, y =  max_y) + theme(axis.title.y = element_blank(),
      axis.text.y  = element_blank(), axis.ticks.y = element_blank())})
    
    output$meanplot <- renderPlot({
      mean_mean = mean(mean_now()$Mean)
      ggplot(data = mean_now(), aes(x=Mean, y=y)) + geom_point(color="firebrick") + geom_vline(xintercept=mean_mean) + annotate("label", label = "Mean of means", x = mean_mean, y =  max_y)+ ylim(0, 0.05) + xlim(95, 105) + theme_classic() + theme(axis.title.y = element_blank(),
      axis.text.y  = element_blank(), axis.ticks.y = element_blank()) })
    
    output$sdplot <- renderPlot({
      SE = mean(sd_now()$SD)
      ggplot(data = sd_now(), aes(x=SD, y=y)) + geom_point(color="steelblue") + geom_vline(xintercept=SE) + annotate("label", label = "Std. Error", x = SE, y =  max_y) + ylim(0, 0.05) + xlim(5, 15) + theme_classic() + theme(axis.title.y = element_blank(),
      axis.text.y  = element_blank(), axis.ticks.y = element_blank())})
    
}




shinyApp(ui, server)
